// V5.2 共通版面去除重複標題 + 薪資結算月份防呆
(() => {
  // 上方導覽列已顯示模組名稱與說明，內容區不再重複顯示相同標題。
  if(!document.getElementById('v52-page-head-dedup-style')){
    const layoutStyle=document.createElement('style');
    layoutStyle.id='v52-page-head-dedup-style';
    layoutStyle.textContent=`
      .page-head > div:first-child:has(> h1){display:none!important}
      .page-head:has(> div:first-child:only-child > h1){display:none!important}
      .page-head:has(> .actions){justify-content:flex-end!important;align-items:center!important}
    `;
    document.head.appendChild(layoutStyle);
  }

  function payrollCurrentMonth(){
    const now=new Date();
    return `${now.getFullYear()}-${String(now.getMonth()+1).padStart(2,'0')}`;
  }

  function payrollSelectedMonthClosed(){
    const selected=String(state.overtimeMonth||'');
    return /^\d{4}-\d{2}$/.test(selected) && selected < payrollCurrentMonth();
  }

  function applyPayrollPeriodGuard(){
    const btn=document.getElementById('payrollCloseBtn');
    if(!btn) return;

    const canClose=payrollSelectedMonthClosed();
    const buttonText=canClose?'結算本月加班費':'本月尚未結束，暫不可結算';
    const buttonTitle=canClose?'確認薪資資料後進行月結':'僅能結算已完整結束的月份；當月請於次月1日起辦理結算。';

    btn.disabled=!canClose;
    btn.style.opacity=canClose?'':'0.55';
    btn.style.cursor=canClose?'':'not-allowed';
    if(btn.textContent!==buttonText) btn.textContent=buttonText;
    if(btn.title!==buttonTitle) btn.title=buttonTitle;

    const card=btn.closest('.ot-rule-card');
    if(!card) return;
    let notice=card.querySelector('[data-payroll-period-notice]');

    if(!canClose){
      if(!notice){
        notice=document.createElement('div');
        notice.dataset.payrollPeriodNotice='1';
        notice.className='ot-rule-highlight';
        card.appendChild(notice);
      }
      const [year,month]=String(state.overtimeMonth||'').split('-').map(Number);
      const nextMonth=new Date(year,month,1);
      const noticeText=`此月份尚未完整結束，需於 ${nextMonth.getFullYear()} 年 ${nextMonth.getMonth()+1} 月 1 日起才能進行月結。`;
      if(notice.textContent!==noticeText) notice.textContent=noticeText;
    }else if(notice){
      notice.remove();
    }
  }

  document.addEventListener('click',e=>{
    const btn=e.target.closest?.('#payrollCloseBtn');
    if(!btn || payrollSelectedMonthClosed()) return;
    e.preventDefault();
    e.stopImmediatePropagation();
    toast('此月份尚未完整結束，請於次月1日起再進行薪資結算。','warn');
  },true);


  // 撤銷月結：只有最後已結算月份可撤銷，權限由資料庫再檢查。
  let undoing=false;
  function renderUndoAction(){
    if(state.page!=='overtime'||state.overtimeView!=='payroll')return;
    if(!state.profile||!['admin','organization_manager'].includes(state.profile.role))return;
    const month=String(state.overtimeMonth||'');
    if(!/^\d{4}-\d{2}$/.test(month))return;
    const selected=month+'-01';
    const records=state.payrollSettlements||[];
    const settled=records.find(r=>r.settlement_month===selected);
    const status=[...document.querySelectorAll('#otMain .ot-status.approved')]
      .find(el=>el.closest('.ot-rule-card')?.querySelector('h3')?.textContent.includes('薪資結算報表'));
    if(!settled||!status||document.getElementById('payrollUndoBtn'))return;
    const latest=records.map(r=>r.settlement_month).sort().at(-1);
    const btn=document.createElement('button');
    btn.id='payrollUndoBtn';btn.type='button';btn.className='btn';
    btn.textContent='↶ 撤銷月結';btn.disabled=selected!==latest;
    btn.style.marginLeft='8px';
    btn.title=btn.disabled?'只能撤銷最後一個已結算月份':'撤銷本月月結並重新開放加班及換休申請';
    btn.addEventListener('click',async()=>{
      if(undoing||btn.disabled)return;
      const [year,monthNum]=selected.split('-').map(Number);
      const note='確定撤銷民國'+(year-1911)+'年'+monthNum+'月的薪資月結嗎？\n\n撤銷後，該月份恢復未結算，可依原有申請及審核流程修正，再重新結算。\n\n若薪資已實際發放，請先確認後續薪資處理方式。';
      if(!confirm(note))return;
      undoing=true;btn.disabled=true;showLoading(true);
      try{
        const {error}=await db.rpc('undo_payroll_month_settlement',{p_settlement_month:selected});
        if(error)throw error;
        state.overtimeView='payroll';
        navigate('overtime');
        state.overtimeView='payroll';
        toast('已撤銷月結，該月份恢復未結算','success');
      }catch(err){
        console.error('撤銷月結失敗',err);
        toast('撤銷月結失敗：'+(err?.message||err),'error');
        btn.disabled=false;
      }finally{undoing=false;showLoading(false);}
    });
    status.insertAdjacentElement('afterend',btn);
  }

  // 月結頁面重繪時刷新按鈕、權限與月份限制。
  const observer=new MutationObserver(mutations=>{
    const shouldApply=mutations.some(m=>Array.from(m.addedNodes||[]).some(node=>{
      if(node.nodeType!==1) return false;
      return node.id==='payrollCloseBtn' || node.id==='otMain' || !!node.querySelector?.('#payrollCloseBtn,#otMain');
    }));
    if(shouldApply){applyPayrollPeriodGuard();renderUndoAction();}
  });
  observer.observe(document.body,{childList:true,subtree:true});
  setTimeout(()=>{applyPayrollPeriodGuard();renderUndoAction();},0);
})();
